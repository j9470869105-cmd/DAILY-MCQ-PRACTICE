<script>

const API_URL =
"https://script.google.com/macros/s/AKfycbxThxC3n3WNOFedD3OwERNYL_WXkFugQQnaqCArMUH_depyCGf0i_Tetk7T_KY2PSdy/exec";

let score = 0;

async function loadQuestions(){

    try{

        const response = await fetch(API_URL);
        const questions = await response.json();

        const container =
        document.getElementById("mcqContainer");

        container.innerHTML = "";

        questions.forEach((q,index)=>{

            const card =
            document.createElement("div");

            card.className = "card";

            card.innerHTML = `
                <div class="question">
                    ${index+1}. ${q.question}
                </div>

                <div class="option" data-option="A">
                    A. ${q.optionA}
                </div>

                <div class="option" data-option="B">
                    B. ${q.optionB}
                </div>

                <div class="option" data-option="C">
                    C. ${q.optionC}
                </div>

                <div class="option" data-option="D">
                    D. ${q.optionD}
                </div>
            `;

            const options =
            card.querySelectorAll(".option");

            options.forEach(option=>{

                option.addEventListener("click",()=>{

                    options.forEach(op=>{
                        op.classList.add("disabled");
                    });

                    const selected =
                    option.dataset.option;

                    const correct =
                    q.correct;

                    options.forEach(op=>{

                        if(op.dataset.option === correct){
                            op.classList.add("correct");
                        }

                    });

                    if(selected !== correct){

                        option.classList.add("wrong");

                    }else{

                        score++;

                        document.getElementById("score")
                        .innerText = score;
                    }

                });

            });

            container.appendChild(card);

        });

    }catch(error){

        document.getElementById("mcqContainer")
        .innerHTML =
        "<h3>Questions load nahi ho pa rahe.</h3>";

        console.error(error);
    }
}

loadQuestions();

</script>
